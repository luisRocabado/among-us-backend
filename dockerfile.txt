FROM python:3.9-slim

# Instalar FFmpeg y dependencias
RUN apt-get update && apt-get install -y ffmpeg && rm -rf /var/lib/apt/lists/*

WORKDIR /app

# Instalar dependencias Python críticas
RUN pip install --no-cache-dir flask flask-cors openai==1.3.7 openai-whisper opencv-python-headless supabase python-dotenv gunicorn

# Copiar código
COPY . .

# Puerto
EXPOSE 10000

# Comando de inicio
CMD ["gunicorn", "app:app", "--bind", "0.0.0.0:10000", "--workers", "1", "--timeout", "1800"]
